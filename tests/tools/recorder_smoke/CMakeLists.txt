cmake_minimum_required(VERSION 3.14)
set(CMAKE_CUDA_COMPILER /usr/local/cuda/bin/nvcc)
set(CMAKE_CUDA_ARCHITECTURES 86)
project(smoke CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_BUILD_TYPE RelWithDebInfo)
find_package(Qt5 REQUIRED COMPONENTS Core)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED libavformat libavcodec libavutil)
set(R ${CMAKE_CURRENT_SOURCE_DIR}/../../..)
add_executable(smoke smoke.cpp ${R}/ui/Recording/VideoRecorder.cpp ${R}/ui/Recording/VideoRecorder.hpp ${R}/cuda/frame_convert.cu)
target_include_directories(smoke PRIVATE ${R} ${R}/include ${R}/decklink /usr/local/cuda/include ${AV_INCLUDE_DIRS})
target_link_directories(smoke PRIVATE /usr/local/cuda/targets/x86_64-linux/lib)
target_link_libraries(smoke Qt5::Core ${AV_LIBRARIES} cuda cudart)
